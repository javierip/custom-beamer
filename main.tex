\section{Theme options}
\begin{frame}
  \frametitle{Theme options}
  \small
  \begin{description}
    \item[\texttt{listings}] loads \texttt{listings} and defines
      \texttt{\textbackslash lstdefinestyle\{utnbhi\}}. Without it the theme
      pulls in nothing extra.
    \item[\texttt{pagebackground}] restores the pre-2.0 full-page raster
      background instead of the footline logo. The image is letterboxed, so
      the logo keeps its proportions in 16:9.
    \item[\texttt{nologo}] draws no logo, but still reserves its space.
    \item[\texttt{classicgreen}] restores the original \texttt{\#CBD300}
      frame titles (1.63:1 against white --- not recommended).
  \end{description}
  \vfill
  For a single group of frames, \texttt{\textbackslash utnbhinologo} hides the
  logo without changing the text area.
\end{frame}

\begin{frame}
  \frametitle{Emphasis and blocks}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{A block}
        Block colours come from the brand palette instead of Beamer's stock
        blue.
      \end{block}
      \begin{exampleblock}{An example block}
        Every block pair measures at least 4.5:1.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{An alert block}
        \alert{Alerted text} uses a dark brand-compatible red.
      \end{alertblock}
      \vspace{1em}
      Body text stays \texttt{grayUTN}; anything on a tint switches to the
      darker \texttt{grayUTNdark}.
    \end{column}
  \end{columns}
\end{frame}

int sum(const int *a, int n) {
  int total = 0;
  for (int i = 0; i < n; i++)
    total += a[i];
  return total;
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{lstlisting}[language=Python]
def total(values):
    """Sum a sequence."""
    result = 0
    for value in values:
        result += value
    return result
\end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}
